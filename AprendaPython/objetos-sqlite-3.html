    Objetos SQLite em Python - Parte 3

    Aprenda o R de Read: Como Listar e Buscar Pessoas

    Vamos visualizar os conteúdos de duas forma, todos os cadastros e procurar por nome.

    Continuando o projeto, faça isso para vermos todos os cadastros, usando o comando SQL select no listarPessoas:

    
def listarPessoas() -> bool:
    print("=== LISTAGEM COMPLETA DE PESSOAS CADASTRADAS ===")
    try:
        with sqlite3.connect(NOME_BANCO) as db:
            cur = db.cursor()

            cur.execute("select cod, nome, altura from pessoas order by nome")

            primeiro = cur.fetchone()

            if not primeiro:
                print("[AVISO] Nenhuma pessoa foi cadastrada até agora.")

                return True

            print("=" * 45)
            print(f"{'CÓDIGO':<7} {'NOME':<28} {'ALTURA':>8}")
            print(f"-" * 45)

            vCod, vNome, vAltura = primeiro # Retorna uma coleção
            print(f"{vCod:<7} {vNome:<28} {vAltura:>6}m")

            for c, n, a in cur:
                print(f"{c:<7} {n:<28} {a:>6}m")

            print("=" * 45)

            return True
    except sqlite3.Error as erro:
        print(f"[ERRO] Falha ao acessar o banco: {erro}!")

        return False

    E para vermos apenas os cadastros com uma palavra-chave ou parte dela no nome, alteramos a função buscarPessoaPorNome assim:

    
def buscarPessoaPorNome() -> bool:
    print("=== BUSCA DE PESSOA POR NOME ===")

    termo = str(input("Digite o nome (ou parte dele) para buscar: ")).strip()

    if not termo:
        print("[AVISO] Nenhum termo foi informado!")

        return False

    try:
        with sqlite3.connect(NOME_BANCO) as db:
            cur = db.cursor()

            cur.execute("select cod, nome, altura from pessoas where nome like ? order by nome", (f"%{termo}%",))

            primeiro = cur.fetchone()

            if not primeiro:
                print("[AVISO] Nenhuma pessoa foi cadastrada até agora.")

                return True

            print("=" * 45)
            print(f"{'CÓDIGO':<7} {'NOME':<28} {'ALTURA':>8}")
            print(f"-" * 45)

            vCod, vNome, vAltura = primeiro # Retorna uma coleção
            print(f"{vCod:<7} {vNome:<28} {vAltura:>6}m")

            for c, n, a in cur:
                print(f"{c:<7} {n:<28} {a:>6}m")

            print("=" * 45)

            return True
    except sqlite3.Error as erro:
        print(f"[ERRO] Falha ao buscar no banco: {erro}!")

        return False

    PS: Podemos usar a biblioteca rich para mostrar os cadastros numa tabela. Lembre-se também de que a porcentagem no SQL é um curinga que significa de zero a infinitos caracteres antes e após o termo que fica entre eles. E o SQLite difere letras acentuadas, mas não difere maiúsculas e minúsculas.
