    Exercícios - Parte 13

    Mostre uma mensagem caso não seja encontrado nenhum registro, trate as datas inválidas e vazias, mostre a data formatada corretamente e mostre a idade real da pessoa:

    
import sqlite3
from datetime import date

busca = str(input("Quem você está procurando? ")).strip()
hoje = date.today()

try:
    with sqlite3.connect("meubanco.db3") as banco:
        banco.row_factory = sqlite3.Row
        cur = banco.cursor()

        cur.execute("select * from amigos where nome like ?", (f"%{busca}%",)) # Não esqueça da vírgula no final

        total = 0

        for total, amigo in enumerate(cur, start = 1):
            txtNasc = amigo["nascimento"]
            infoNasc = ""

            if not txtNasc:
                infoNasc = "sem data de nascimento"
            else:
                try:
                    data = date.fromisoformat(str(txtNasc)) # Isso formata a data igual a gente está acostumado.
                    jaPassou = (hoje.month, hoje.day) >= (data.month, data.day) # Comparação de tuplas, na ordem mês e dia
                    idade = hoje.year - data.year - (0 if jaPassou else 1) # Analisa se no ano atual passou aniversário
                    infoNasc = f"nasceu em {data:%d/%m/%Y} e tem {idade} anos"
                except ValueError:
                    infoNasc = "data inválida"

            print(f"{amigo['nome']} {infoNasc}.")

        if total == 0:
            print(f"Não encontrei ninguém com \"{busca}\" no nome.")
        elif total == 1:
            print(f"Encontrei apenas um amigo com \"{busca}\".")
        else:
            print(f"Ao todo encontrei {total} amigos com \"{busca}\".")
except sqlite3.Error as erro:
    print(f"Falha no banco: {erro}")
except Exception as erro:
    print(f"Algo deu errado {erro}")
